file(GLOB PMIDI_SOURCES *.c)
add_executable(pmidi ${PMIDI_SOURCES})
target_include_directories(pmidi PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(pmidi PRIVATE turbosynth)

if(CMAKE_SYSTEM_NAME STREQUAL "Windows")
	target_sources(pmidi PRIVATE pmidi.rc)
else()
	target_link_libraries(pmidi PRIVATE m pthread)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
	target_link_libraries(pmidi PRIVATE dl)
endif()

if(MW_DIR AND NOT MW_INCLUDE_DIR)
	set(MW_INCLUDE_DIR ${MW_DIR}/include)
endif()
if(MW_DIR AND NOT MW_LIBRARY_DIR)
	set(MW_LIBRARY_DIR ${MW_DIR}/lib ${MW_DIR}/src)
endif()

if(MW_INCLUDE_DIR AND MW_LIBRARY_DIR)
	set(MW ON)
else()
	find_library(MW Mw PATHS ${MW_LIBRARY_DIR})
endif()

if(MW)
	target_compile_definitions(pmidi PRIVATE HAS_MW)

	target_include_directories(pmidi PRIVATE ${MW_INCLUDE_DIR})
	target_link_directories(pmidi PRIVATE ${MW_LIBRARY_DIR})
	target_link_libraries(pmidi PRIVATE Mw)
endif()

include(GNUInstallDirs)
configure_file(config.h.in config.h)
